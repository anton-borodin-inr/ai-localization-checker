def sha(data):
    return hashlib.sha256(data).hexdigest()


def write_json(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n")


def translation(language, key, value):
    relative = f"android/assets/jsons/translations/{language}.properties"
    text = (FORK / relative).read_text()
    matches = [line for line in text.splitlines() if line.startswith(key + " = ")]
    assert len(matches) <= 1, (language, key)
    return {
        "file": relative,
        "key": key,
        "original": matches[0].split(" = ", 1)[1] if matches else None,
        "mutated": value,
        "operation": "remove" if value is None else "replace" if matches else "add",
    }


def variant(name, screen, candidates, de, en, note):
    mutations = [translation("German", *de)]
    if en is not None:
        mutations.append(translation("English", *en))
    return {
        "id": name,
        "screen": screen,
        "candidates": candidates,
        "languages": ["de", "en"] if en is not None else ["de"],
        "mutations": mutations,
        "design_note": note,
        "observed": None,
    }
