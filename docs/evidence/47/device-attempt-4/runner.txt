from pathlib import Path
ROOT=Path(__file__).resolve().parents[2];CHECKER=ROOT/'work/checker';OUT=ROOT/'work/device-execution-4'

PLAN_SHA='4d8b08cef5077e62551813fe4e55678f352ab6a2ad4be15fd720c23fbf32828a'

def control_flow(ops):
    """Never retry an APK write; restore target once after any begun old install."""
    errors=[];began=False
    def attempt(phase,action):
        try:action();return True
        except BaseException as e:
            errors.append({'phase':phase,'exception':type(e).__name__,'error':str(e),'traceback':traceback.format_exc()});return False
    try:
        if attempt('preflight',ops.preflight):
            old_ok=False;target_ok=False
            try:
                began=True
                old_ok=attempt('reference_phase',lambda:(ops.install_old(),ops.verify_old(),ops.capture_old()))
            finally:
                if began:target_ok=attempt('target_restoration',lambda:(ops.install_target(),ops.verify_target()))
            if old_ok and target_ok:attempt('target_capture',ops.capture_target)
    except BaseException as e:
        errors.append({'phase':'control_flow_interruption','exception':type(e).__name__,'error':str(e),'traceback':traceback.format_exc()})
    finally:
        attempt('final_preservation',lambda:ops.finalize(began))
    return errors

class AuditDevice(Device):
    def __init__(self,*args):super().__init__(*args);self.commands=[]
    def run(self,*args,**kw):
        row={'arguments':list(args),'started_monotonic':time.monotonic()};self.commands.append(row)
        try:
            data=super().run(*args,**kw);row.update(status='returned',size_bytes=len(data),sha256=digest(data));return data
        except BaseException as e:row.update(status='failed',exception=type(e).__name__);raise
        finally:
            row['finished_monotonic']=time.monotonic();(OUT/'commands.json').write_text(json.dumps(self.commands,indent=2)+'\n')

class Collect:
    def __init__(self,log,lines):self.log=log;self.lines=lines
    def next_line(self,deadline):
        line=self.log.next_line(deadline);self.lines.append(line);return line

class Operations:
    def __init__(self,p,original,old_inputs,target_inputs):
        self.p=p;self.original=original;self.old_inputs=old_inputs;self.target_inputs=target_inputs;self.rows=[];self.data_checks=[];self.source_verified=False
        self.d=AuditDevice(Path(p['requests'][0]['command'][0]),p['device']['serial'])
    def hook_inventory(self):
        result={}
        for r in self.p['existing_hook_outputs_to_verify']:
            data=self.d.run('exec-out','run-as','com.unciv.app','cat',r['file']);result[r['file']]=digest(data)
        return result
    def data_check(self,label,backup=False):
        d=self.d
        try:
            d.root();d.force_stop()
            if backup:
                folder=CHECKER/'data/issue-47/control-4-preflight';folder.mkdir(exist_ok=False);snap=backup_device(d,self.target_inputs,folder)
                data=snap['original'];stat=snap['stat'];saves=snap['saves']
            else:data=d.read_file(SETTINGS_PATH);stat=d.stat(SETTINGS_PATH);saves=save_inventory(read_saves(d))
            hooks=self.hook_inventory()
            row={'phase':label,'settings_sha256':digest(data),'settings_stat':stat,'saves':saves,'hook_files':hooks};self.data_checks.append(row)
            (OUT/'data-checks.json').write_text(json.dumps(self.data_checks,indent=2)+'\n')
            assert data==self.original and stat==self.p['settings']['settings_stat_uid_gid_mode_size'],'Original settings pin changed'
            assert saves==self.p['saves'],'Save changed; no save overwrite authorized'
            assert hooks=={r['file']:r['sha256'] for r in self.p['existing_hook_outputs_to_verify']},'Existing hook output changed; no overwrite authorized'
        finally:d.unroot()
    def preflight(self):
        d=self.d;assert check_build(d,self.target_inputs)['apk_sha256']==self.p['source_installed']['apk_sha256']
        assert d.shell('getprop','ro.build.version.sdk').strip()=='34' and d.shell('getprop','ro.product.cpu.abi').strip()=='arm64-v8a'
        assert d.run('emu','avd','name').decode().splitlines()[0]=='unciv_api34'
        self.source_verified=True
        self.data_check('preflight',backup=True)
    def install(self,index):
        operation=self.p['install_operations'][index];name=operation['name'];marker=OUT/f'{name}-started.json';assert not marker.exists(),'No install retry'
        self.d.force_stop();marker.write_text(json.dumps({'command':operation['command'],'started_epoch':time.time()},indent=2)+'\n')
        started=time.monotonic()
        try:
            r=subprocess.run(operation['command'],capture_output=True,timeout=120)
            (OUT/f'{name}-stdout.txt').write_bytes(r.stdout);(OUT/f'{name}-stderr.txt').write_bytes(r.stderr)
            result={'command':operation['command'],'exit_code':r.returncode,'elapsed_seconds':time.monotonic()-started,'stdout':r.stdout.decode(errors='replace'),'stderr':r.stderr.decode(errors='replace')}
            (OUT/f'{name}-result.json').write_text(json.dumps(result,indent=2)+'\n')
            assert r.returncode==0 and 'Success' in result['stdout'],'PackageManager update failed; no retry'
        except subprocess.TimeoutExpired as e:
            (OUT/f'{name}-result.json').write_text(json.dumps({'command':operation['command'],'status':'timeout','elapsed_seconds':time.monotonic()-started},indent=2)+'\n');raise
    def install_old(self):self.install(0)
    def install_target(self):self.install(1)
    def verify_old(self):
        observed=check_build(self.d,self.old_inputs);(OUT/'old-build.json').write_text(json.dumps(observed,indent=2)+'\n');self.data_check('after-reference-update')
    def verify_target(self):
        observed=check_build(self.d,self.target_inputs);(OUT/'restored-build.json').write_text(json.dumps(observed,indent=2)+'\n');self.data_check('after-target-update')
    def capture(self,index):
        r=self.p['requests'][index];d=self.d;d.check_cutout();assert d.build()==r['build_version_code'];d.force_stop()
        row={'check':r['check'],'request':r['request'],'version_code':r['build_version_code'],'lines':[],'images':[]};self.rows.append(row)
        try:
            with Logcat(d) as log:
                reader=Collect(log,row['lines']);started=time.monotonic();row['am_start']=d.run(*r['command'][3:]).decode(errors='replace')
                assert 'Starting: Intent' in row['am_start'] and 'Error:' not in row['am_start']
                state=json.loads(base64.b64decode(r['command'][r['command'].index('l10n_state_b64')+1]))
                row['readiness']=wait_terminal(reader,'tech-picker',r['request'],set(state['scroll'])|set(state['scroll_to']),started+self.p['timeout_seconds_per_request'],1);row['terminal_elapsed_seconds']=time.monotonic()-started
                assert 'com.unciv.app' in d.foreground()
                for n,name in enumerate(r['host_png_files']):
                    if n:time.sleep(2)
                    png=d.screenshot();assert png_size(png)==(2400,1080);(OUT/name).write_bytes(png);row['images'].append({'file':name,'sha256':digest(png),'rgba_sha256':rgba_digest(png)})
                    (OUT/f'{r["check"]}-gfxinfo-{n}.txt').write_text(d.shell('dumpsys','gfxinfo','com.unciv.app','framestats'))
                    layers=d.shell('dumpsys','SurfaceFlinger','--list');(OUT/f'{r["check"]}-layers-{n}.txt').write_text(layers)
                    selected=[x for x in layers.splitlines() if 'com.unciv.app/com.unciv.app.AndroidLauncher' in x and 'BLAST' in x]
                    if selected:(OUT/f'{r["check"]}-latency-{n}.txt').write_text(d.shell('dumpsys SurfaceFlinger --latency '+shlex.quote(selected[-1]),allow_fail=True))
                until=time.monotonic()+2
                while True:
                    try:reader.next_line(until)
                    except TimeoutError:break
            events=[EVENT.search(line) for line in row['lines']];events=[e for e in events if e and e[3]==r['request']]
            assert [e[1] for e in events if e[1] in ('READY','ERROR')]==['READY'] and sum(e[1]=='PANES' for e in events)==1 and not any(e[1]=='DUMP' for e in events)
            row['status']='observed';self.data_check('after-'+r['check']+'-capture');print(r['check'],'observed',flush=True)
        finally:(OUT/'captures.json').write_text(json.dumps(self.rows,indent=2)+'\n')
    def capture_old(self):self.capture(0)
    def capture_target(self):self.capture(1)
    def finalize(self,began):
        if not began and not self.source_verified:
            (OUT/'final-preservation.json').write_text(json.dumps({'status':'not_run_source_preflight_rejected','apk_update_phase_began':False,'device_cleanup_commands':0,'reason':'Do not root or force-stop an unrelated/changed device after source preflight rejection'},indent=2)+'\n');return
        d=self.d;result={'apk_update_phase_began':began,'conditional_settings_restore_used':False}
        try:
            d.root();d.force_stop();result.update(installed_version_code=d.build(),app_running=bool(d.pid()))
            paths=d.shell('pm','path','com.unciv.app').strip().splitlines();assert len(paths)==1 and paths[0].startswith('package:')
            result['installed_apk_sha256']=d.shell('sha256sum '+shlex.quote(paths[0].removeprefix('package:'))).split()[0]
            current=d.read_file(SETTINGS_PATH);stat=d.stat(SETTINGS_PATH);saves=save_inventory(read_saves(d))
            result.update(settings_before_sha256=digest(current),settings_before_stat=stat,saves=saves)
            if current!=self.original or stat!=self.p['settings']['settings_stat_uid_gid_mode_size']:
                assert began,'Preflight settings mismatch: do not overwrite'
                marker=OUT/'conditional-settings-restore.json';assert not marker.exists()
                marker.write_text(json.dumps({'status':'attempted','before_sha256':digest(current),'target_sha256':digest(self.original)},indent=2)+'\n')
                private=OUT/'private';private.mkdir(mode=0o700,exist_ok=True);backup=private/'settings-before-recovery.json';backup.write_bytes(current);backup.chmod(0o600)
                d.write_file(SETTINGS_PATH,self.original);uid,gid,mode,_=self.p['settings']['settings_stat_uid_gid_mode_size'].split();d.shell('chown',uid+':'+gid,SETTINGS_PATH);d.shell('chmod',mode,SETTINGS_PATH);result['conditional_settings_restore_used']=True
            data=d.read_file(SETTINGS_PATH);stat=d.stat(SETTINGS_PATH);hooks=self.hook_inventory();result['hook_files']=hooks
            result.update(settings_sha256=digest(data),settings_stat=stat,settings_match_original=data==self.original and stat==self.p['settings']['settings_stat_uid_gid_mode_size'],saves_match_original=saves==self.p['saves'],prior_hook_outputs_unchanged=hooks=={r['file']:r['sha256'] for r in self.p['existing_hook_outputs_to_verify']},installed_version_code=d.build(),app_running=bool(d.pid()))
        finally:
            try:
                d.unroot();result['adbd']=d.shell('whoami').strip()
            except BaseException as e:
                result['unroot_error']=str(e);raise
            finally:(OUT/'final-preservation.json').write_text(json.dumps(result,indent=2)+'\n')
        assert result['settings_match_original'] and result['saves_match_original'] and result['prior_hook_outputs_unchanged']
        assert result['installed_version_code']=='1271048' and result['installed_apk_sha256']==self.p['final_target']['apk_sha256']
        assert not result['app_running'] and result['adbd']=='shell'

def main():
    raw=(ROOT/'outputs/issue-47-device-plan-4.json').read_bytes();assert digest(raw)==PLAN_SHA;p=json.loads(raw)
    assert len(p['requests'])==2 and len(p['install_operations'])==2 and not p['normal_settings_writes'] and not p['save_writes'] and not p['new_hook_output_files']
    for b in (p['reference_build'],p['final_target']):assert digest(Path(b['apk']).read_bytes())==b['apk_sha256']
    old=Inputs(Path(p['reference_build']['manifest']));target=Inputs(Path(p['final_target']['manifest']));assert old.sha256==p['reference_build']['manifest_sha256'] and target.sha256==p['final_target']['manifest_sha256']
    for r in p['requests']:
        assert r['protocol_version']==1 and 'l10n_protocol' not in r['command'] and re.fullmatch(r'[A-Za-z0-9._-]{1,64}',r['request'])
        assert digest(base64.b64decode(r['command'][r['command'].index('l10n_state_b64')+1],validate=True))==r['state_sha256']
    assert p['install_operations'][0]['command'][3:6]==['install','-r','-d'] and p['install_operations'][1]['command'][3:5]==['install','-r']
    assert len(p['existing_hook_outputs_to_verify'])==45 and len({r['file'] for r in p['existing_hook_outputs_to_verify']})==45
    assert all(re.fullmatch(r'files/l10n-hook/[A-Za-z0-9._-]{1,64}\.json',r['file']) for r in p['existing_hook_outputs_to_verify'])
    original=Path(p['conditional_settings_recovery']['payload_source']).read_bytes();assert digest(original)==p['settings']['settings_sha256']
    assert not subprocess.check_output(['git','diff',p['checker_runtime_commit'],'--','src'],cwd=CHECKER)
    if '--validate-only' in sys.argv:
        print(json.dumps({'status':'host_validated_no_device_access','plan_sha256':PLAN_SHA,'apk_updates_max':2,'requests':2,'new_hook_files':0,'normal_settings_writes':0,'prior_hook_files_to_verify':45},indent=2));return 0
    receipt=json.loads((OUT/'authorization.json').read_bytes());assert receipt['approved_by']=='Anton' and receipt['plan_sha256']==PLAN_SHA
    assert not (OUT/'started.json').exists();(OUT/'started.json').write_text(json.dumps({'started_epoch':time.time(),'plan_sha256':PLAN_SHA},indent=2)+'\n')
    errors=control_flow(Operations(p,original,old,target));(OUT/'result.json').write_text(json.dumps({'status':'failed' if errors else 'completed','errors':errors,'provider_requests':0,'spend_usd':0},indent=2)+'\n')
    return 1 if errors else 0
if __name__=='__main__':raise SystemExit(main())
